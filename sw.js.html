1. Cấu Trúc 3 File Cần Đặt Cùng Thư Mục

File 1: sw.js (Service Worker chạy ngầm) Bắt buộc

Đây là tiến trình nền độc lập với tab duyệt web. Khi điện thoại tắt màn hình, Service Worker giữ quyền đánh thức thiết bị và phát thông báo. 

self.addEventListener('install', (e) => self.skipWaiting());
self.addEventListener('activate', (e) => e.waitUntil(self.clients.claim()));

// Khi bấm vào thông báo trên màn hình khóa -> Tự động bật mở trang web
self.addEventListener('notificationclick', (event) => {
    event.notification.close();
    event.waitUntil(
        clients.matchAll({ type: 'window', includeUncontrolled: true }).then((clientList) => {
            for (const client of clientList) {
                if (client.url && 'focus' in client) return client.focus();
            }
            if (clients.openWindow) return clients.openWindow('./');
        })
    );
});
